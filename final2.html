import polars as pl

from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeClassifier
from sklearn import metrics
url = "https://raw.githubusercontent.com/byuidatascience/data4dwellings/master/data-raw/dwellings_ml/dwellings_ml.csv"
dwellings = pl.read_csv(url)
dwellings = dwellings.with_columns(
    (pl.col("nocars") > 2)
    .cast(pl.Int64)
    .alias("many_cars")
)
X = dwellings.drop([
    "parcel",
    "nocars",
    "many_cars"
])

y = dwellings.get_column("many_cars")

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.30,
    random_state=2021
)


model = DecisionTreeClassifier(
    random_state=2021
)

model.fit(
    X_train,
    y_train
)


predictions = model.predict(
    X_test
)

accuracy = metrics.accuracy_score(
    y_test,
    predictions
)

print(round(accuracy, 2))
0.88

import polars as pl
from lets_plot import *

LetsPlot.setup_html(isolated_frame=True)

url = (
    "http://raw.githubusercontent.com/byuidatascience/data_fivethirtyeight_starwars/master/survey_wide_all.csv"
)

starwars = pl.read_csv(url)


fan_starwars = (
    starwars
    .select([
        "gender",
        "fan_yn"
    ])
    .filter(
        pl.col("gender").is_not_null()
    )
    .with_columns(
        (
            pl.col("fan_yn")
            == "Yes"
        )
        .alias("a_fan")
    )
)

fan_proportion = (
    fan_starwars
    .group_by("gender")
    .agg(
        pl.col("a_fan")
        .mean()
        .alias("proportion")
    )
    .sort("gender")
)

print(
    fan_proportion.with_columns(
        pl.col("proportion").round(2)
    )
)
(
    ggplot(
        fan_proportion,
        aes(
            x="gender",
            y="proportion",
            fill="gender"
        )
    )
    + geom_bar(stat="identity")
    + labs(
        title="A real fan of Star Wars by Gender",
        x="Gender",
        y="Proportion",
        fill="Gender"
    )
    
    + theme_light()
)
